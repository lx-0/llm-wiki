## What this does

`append_history`:
```python
def append_history(event_type: str, **fields) -> None:
    """Append one JSON line to STATE_DIR/history.jsonl. Atomic per-line write."""
    HISTORY_FILE = STATE_DIR / "history.jsonl"
    HISTORY_FILE.parent.mkdir(parents=True, exist_ok=True)
    payload = {"ts": now_iso(), "type": event_type, **fields}
    with HISTORY_FILE.open("a", encoding="utf-8") as fh:
        fh.write(json.dumps(payload, separators=(",", ":")) + "\n")
```

`read_history`:
```python
def read_history(limit: int | None = None) -> list[dict]:
    """Read history.jsonl, skip malformed lines silently. Newest-last."""
    HISTORY_FILE = STATE_DIR / "history.jsonl"
    if not HISTORY_FILE.exists():
        return []
    events = []
    for line in HISTORY_FILE.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line:
            continue
        try:
            events.append(json.loads(line))
        except json.JSONDecodeError:
            continue  # skip bad lines, don't crash
    if limit is not None:
        return events[-limit:]
    return events
```
