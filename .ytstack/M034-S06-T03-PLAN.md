## What this does

```python
# collectors/__init__.py
MODULES = {"calendar": "collectors.calendar_collector", "email": "collectors.email_collector", …}

# collectors/base.py
def get_collector(name):
    module = MODULES.get(name)          # (the table imported lazily to avoid a cycle)
    if module is None: return None
    importlib.import_module(module)     # @register runs
    cls = _COLLECTORS.get(name)
    if cls is None: raise ValueError(f"{module} registers no collector named {name!r}")
    return cls()
def all_collectors(): return [get_collector(name) for name in MODULES]
```
