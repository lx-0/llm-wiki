## What this does

```python
# Pseudocode:
def main():
    for source_name in ("voice", "pictures"):
        inbox_attr = "voice_inbox" if source_name == "voice" else "picture_inbox"
        inbox_str = getattr(CONFIG.personal, inbox_attr, "").strip()
        if not inbox_str:
            log.info("%s: skipped (CONFIG.personal.%s unset)", source_name, inbox_attr)
            continue
        inbox = Path(inbox_str).expanduser()
        legacy = inbox / ".processed"
        if not legacy.exists():
            log.info("%s: nothing to migrate (%s does not exist)", source_name, legacy)
            continue
        dest = RAW_DIR / "inbox-mobile" / source_name
        dest.mkdir(parents=True, exist_ok=True)
        moved = 0
        for src in legacy.iterdir():
            if src.is_dir():
                continue  # don't recurse — defensive
            dst = dest / src.name
            if dst.exists():
                mtime = datetime.fromtimestamp(src.stat().st_mtime).strftime("%Y%m%dT%H%M%S")
                dst = dest / f"{src.stem}-{mtime}{src.suffix}"
            shutil.move(str(src), str(dst))
            moved += 1
        log.info("%s: moved %d files → %s", source_name, moved, dest)
        try:
            legacy.rmdir()
            log.info("%s: removed empty %s", source_name, legacy)
        except OSError as exc:
            log.warning("%s: rmdir failed (non-empty?): %s", source_name, exc)
```

Setup: re-uses standard logging + `sys.path.insert(0, scripts/)` pattern. CLI via `argparse` with `--dry-run`.
